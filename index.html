const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server, { cors: { origin: "*" } }); app.use(express.static(__dirname)); const MAP_SIZE = 2500; let players = {}; let foods = []; // Генерация еды на сервере for (let i = 0; i < 200; i++) { foods.push({ id: i, x: Math.random() * MAP_SIZE, y: Math.random() * MAP_SIZE, color: ['#ff4b4b', '#ffc832', '#3296ff', '#e650fa', '#00e682'][Math.floor(Math.random() * 5)], size: Math.random() * 3 + 5 }); } io.on('connection', (socket) => { // Новый игрок подключается players[socket.id] = { id: socket.id, head: { x: Math.random() * (MAP_SIZE - 400) + 200, y: Math.random() * (MAP_SIZE - 400) + 200 }, segments: [], color: `hsl(${Math.floor(Math.random() * 360)}, 80%, 50%)`, angle: 0, score: 0 }; // Отправляем игроку начальное состояние socket.emit('init', { id: socket.id, foods: foods }); // Обновление координат от игрока socket.on('player-update', (data) => { if (players[socket.id]) { players[socket.id].head = data.head; players[socket.id].segments = data.segments; players[socket.id].angle = data.angle; players[socket.id].score = data.score; } }); // Чат socket.on('chat-message', (msg) => { io.emit('chat-message', { sender: socket.id.substr(0, 5), text: msg }); }); // Поедание еды socket.on('food-eaten', (foodId) => { if (foods[foodId]) { foods[foodId].x = Math.random() * MAP_SIZE; foods[foodId].y = Math.random() * MAP_SIZE; io.emit('food-respawn', { id: foodId, x: foods[foodId].x, y: foods[foodId].y }); } }); // Автоматическое удаление при отключении socket.on('disconnect', () => { delete players[socket.id]; io.emit('player-left', socket.id); }); }); // Отправка состояния всем игрокам 20 раз в секунду setInterval(() => { io.emit('state-update', players); }, 50); const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
